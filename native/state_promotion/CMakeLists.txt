cmake_minimum_required(VERSION 3.24)
project(bb_state_promotion LANGUAGES C CXX)
find_package(LLVM 21.1.8 EXACT REQUIRED CONFIG)
# Relocate the official Windows SDK's absolute build-machine DIA dependency.
if(WIN32 AND LLVM_ENABLE_DIA_SDK AND TARGET LLVMDebugInfoPDB)
  find_library(REMILL_DIA_GUIDS NAMES diaguids
    PATHS "$ENV{VSINSTALLDIR}/DIA SDK/lib/amd64" NO_DEFAULT_PATH REQUIRED)
  get_target_property(_remill_pdb_deps LLVMDebugInfoPDB INTERFACE_LINK_LIBRARIES)
  set(_remill_pdb_relocated)
  foreach(_dep IN LISTS _remill_pdb_deps)
    if(_dep MATCHES "diaguids[.]lib$")
      list(APPEND _remill_pdb_relocated "${REMILL_DIA_GUIDS}")
    else()
      list(APPEND _remill_pdb_relocated "${_dep}")
    endif()
  endforeach()
  set_property(TARGET LLVMDebugInfoPDB PROPERTY INTERFACE_LINK_LIBRARIES "${_remill_pdb_relocated}")
endif()
add_executable(bb-state-promotion main.cpp)
target_include_directories(bb-state-promotion PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(bb-state-promotion PRIVATE ${LLVM_DEFINITIONS})
target_compile_features(bb-state-promotion PRIVATE cxx_std_17)
llvm_map_components_to_libnames(llvm_libs core irreader analysis support)
target_link_libraries(bb-state-promotion PRIVATE ${llvm_libs})
